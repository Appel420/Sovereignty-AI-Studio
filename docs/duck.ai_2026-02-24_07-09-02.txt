function loadQueue(){
  try{ return JSON.parse(localStorage.getItem(QUEUE_KEY) || '[]'); } catch { return []; }

function dequeue(){

async function fetchWithMode(input, init = {}){
  const mode = getMode();
  if(mode === 'ghost'){
    log('Blocked network request in Ghost mode.');
    const err = new Error('Network disabled in Ghost mode');
    err.code = 'GHOST_NETWORK_BLOCK';
    throw err;
  }
  if(mode === 'live'){
    log('Performing LIVE fetch: ' + input);
    return fetch(input, init);
  }
  // Hybrid: enqueue and return a promise that resolves when request completes
  if(mode === 'hybrid'){
    log('Queueing request in Hybrid mode: ' + input);
    const payload = { input, init, timestamp: Date.now() };
    enqueue(payload);
    // Return a resolved-ish promise indicating queued
    return Promise.resolve({ queued: true });
  }
}

/ Flush queue: attempt in-order /
async function flushQueue(){
  const q = loadQueue();
  if(q.length === 0){ log('Queue empty.'); return; }
  log('Flushing queue (' + q.length + ')...');
  // send sequentially
  while(true){
    const item = dequeue();
    if(!item) break;
    try{
      log('Retrying: ' + item.input);
      const res = await fetch(item.input, item.init);
      log('Flushed: ' + item.input + ' — status ' + res.status);
    } catch (e){
      log('Flush failed for ' + item.input + ' — re-enqueueing and aborting flush.');
      enqueue(item); // put back
      break;
    }
  }
}

        // Compute Merkle Root function
        async function computeMerkleRoot(hashes) {
            let layer = [...hashes];
            while (layer.length > 1) {
                const next = [];
                for (let i = 0; i < layer.length; i += 2) {
                    const left = layer[i];
                    const right = i + 1 < layer.length ? layer[i + 1] : left;
                    next.push(await computeBLAKE3(left + right));

                layer = next;
