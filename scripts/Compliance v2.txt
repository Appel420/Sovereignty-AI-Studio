from agip_core.domain.control_mapping import ControlMapping
from agip_core.domain.evidence import Evidence


class ControlMappingError(Exception):
    pass


class ControlMapper:

    def validate(
        self,
        mapping: ControlMapping,
        evidence_store: dict[str, Evidence]
    ):

        evidence = evidence_store.get(
            mapping.evidence_ref
        )

        if evidence is None:
            raise ControlMappingError(
                "Evidence reference missing"
            )


        if evidence.verification_status != "verified":
            raise ControlMappingError(
                "Evidence is not verified"
            )


        if mapping.status == "rejected":
            raise ControlMappingError(
                "Rejected mapping"
            )


        return mapping

3. Metric Validator
agip_core/core/governance/metric_validator.py
from agip_core.domain.metric import MetricLineage


class MetricLineageError(Exception):
    pass


class MetricValidator:

    def validate(
        self,
        metric: MetricLineage,
        evidence_store: dict,
        mappings: dict,
        alignments: dict
    ):

        if not metric.evidence_chain:
            raise MetricLineageError(
                "Metric has no evidence chain"
            )


        for evidence_id in metric.evidence_chain:

            evidence = evidence_store.get(
                evidence_id
            )

            if evidence is None:
                raise MetricLineageError(
                    f"Missing evidence: {evidence_id}"
                )


            if evidence.verification_status != "verified":
                raise MetricLineageError(
                    "Metric references unverified evidence"
                )


        for mapping_id in metric.derived_from:

            if mapping_id not in mappings:
                raise MetricLineageError(
                    f"Missing control mapping: {mapping_id}"
                )


        for framework in metric.framework_refs:

            if framework not in alignments:
                raise MetricLineageError(
                    f"Missing framework alignment: {framework}"
                )


        return metric

from agip_core.domain.metric import MetricLineage


class MetricLineageError(Exception):
    pass


class MetricValidator:

    def validate(
        self,
        metric: MetricLineage,
        evidence_store: dict,
        mappings: dict,
        alignments: dict
    ) -> MetricLineage:


        if metric.verification_status != "verified":

            raise MetricLineageError(
                "Unverified metrics cannot be published"
            )


        if len(metric.derived_from) == 0:

            raise MetricLineageError(
                "Metric requires control mapping lineage"
            )


        for mapping_id in metric.derived_from:

            if mapping_id not in mappings:

                raise MetricLineageError(
                    f"Missing control mapping: {mapping_id}"
                )


        if len(metric.evidence_chain) == 0:

            raise MetricLineageError(
                "Metric requires evidence chain"
            )


        for evidence_id in metric.evidence_chain:

            evidence = evidence_store.get(
                evidence_id
            )

            if evidence is None:

                raise MetricLineageError(
                    f"Missing evidence: {evidence_id}"
                )


            if evidence.verification_status != "verified":

                raise MetricLineageError(
                    f"Evidence not verified: {evidence_id}"
                )


        for framework in metric.framework_refs:

            if framework not in alignments:

                raise MetricLineageError(
                    f"Missing framework alignment: {framework}"
                )


        return metric

```python
import json
import sys
from pathlib import Path

def load_policy(path: str) -> dict:
    return json.loads(Path(path).read_text(encoding="utf-8"))

def scan_file(path: Path, patterns: list[str]) -> list[str]:
    text = path.read_text(encoding="utf-8", errors="ignore")
    hits = []
    for pattern in patterns:
        if pattern in text:
            hits.append(pattern)
    return hits

def main() -> int:
    policy_path = sys.argv[1]
    root_path = Path(sys.argv[2]) if len(sys.argv) > 2 else Path(".")
    policy = load_policy(policy_path)

    protected = policy["protected_paths"]
    patterns = [p for block in policy["forbidden_patterns"] for p in block["patterns"]]

    violations = []
    for protected_dir in protected:
        base = root_path / protected_dir
        if not base.exists():
            continue
        for file_path in base.rglob("*"):
            if file_path.is_file():
                hits = scan_file(file_path, patterns)
                if hits:
                    violations.append(f"{file_path}: {', '.join(hits)}")

    if violations:
        print("ZERO_TOLERANCE_VIOLATIONS")
        for v in violations:
            print(v)
        return 1

    print("ZERO_TOLERANCE_OK")
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```
