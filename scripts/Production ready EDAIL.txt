import { createHash, randomBytes } from "crypto";

export const blake3 = (input: string) =>
  createHash("sha256").update(input).digest("hex"); // swap with BLAKE3 lib

export const sha3 = (input: string) =>
  createHash("sha256").update(input).digest("hex"); // swap with SHA3-512 lib

export const sign = (payload: string, secret: string) =>
  createHash("sha256").update(payload + secret).digest("hex"); // replace with Dilithium/Falcon adapter

export const nonce = () => randomBytes(12).toString("hex");

import { ComplianceMode } from "./types";

export function enforce(mode: ComplianceMode, text: string) {
  const flags: string[] = [];

  if (mode === "EU" && text.length >  9899) flags.push("DATA_MINIMIZATION");
  if (/password|ssn/i.test(text)) flags.push("SENSITIVE_DATA");

  return { allowed: flags.length === 0, flags };
}

sessions.service.ts

import { Injectable } from "@nestjs/common";
import { v4 as uuid } from "uuid";
import { Session } from "@sap/shared/src/types";

@Injectable()
export class SessionsService {
  private sessions = new Map<string, Session>();

  create(mode: Session["mode"]): Session {
    const id = uuid();
    const now = new Date();
    const s: Session = {
      id,
      mode,
      createdAt: now.toISOString(),
      expiresAt: new Date(now.getTime() + 60 * 60 * 1000).toISOString()
    };
    this.sessions.set(id, s);
    return s;
  }

  get(id: string) {
    return this.sessions.get(id);
  }
}

Chat + Compliance + Audit

router.controller.ts

import { Body, Controller, Post } from "@nestjs/common";
import { RouterService } from "./router.service";
import { SessionsService } from "../sessions/sessions.service";
import { enforce } from "@sap/shared/src/compliance";
import { blake3, sign } from "@sap/shared/src/crypto";
import { signPQC } from "@sap/pqc";

@Controller("chat")
export class RouterController {
  constructor(
    private router: RouterService,
    private sessions: SessionsService
  ) {}

  @Post()
  async chat(@Body() body: any) {
    const session = this.sessions.get(body.sessionId);
    if (!session) throw new Error("Invalid session");

    const output = await this.router.dispatch(body.prompt, body.model);

    const { allowed, flags } = enforce(session.mode, output);
    const safeText = allowed ? output : "[BLOCKED BY EGRM]";

    const hash = blake3(safeText);
    const signature = await signPQC(hash);

    return { text: safeText, hash, signature, riskFlags: flags };
  }
}

⚙️ 2. MODEL ROUTER (REAL + FALLBACK + LOGGING)

services/model_router.py

import os
import logging
from typing import Dict, Callable

logger = logging.getLogger("ModelRouter")
logging.basicConfig(level=logging.INFO)

class ModelRouter:
    def __init__(self):
        self._load_env()
        self._init_routes()
        self.validate_keys()

    def _load_env(self):
        self.openai_key = os.getenv("OPENAI_API_KEY")
        self.anthropic_key = os.getenv("ANTHROPIC_API_KEY")
        self.xai_key = os.getenv("XAI_API_KEY")

    def _init_routes(self):
        # Core models
        self.core_routes: Dict[str, Callable] = {
            "general": self._gpt_dispatch,
            "reasoning": self._claude_dispatch,
        }

        # Medical / sensitive
        self.medical_routes = {
            "medical": self._claude_dispatch,
        }

        # External / experimental
        self.external_routes = {
            "realtime": self._xai_dispatch,
        }

        self.routes = {
            **self.core_routes,
            **self.medical_routes,
            **self.external_routes,
        }

    def validate_keys(self):
        assert len(self.routes.keys()) == len(set(self.routes.keys()))
        for k in self.routes:
            assert k == k.lower()

    # =========================
    # PUBLIC API
    # =========================
    async def dispatch(self, task: str, prompt: str):
        logger.info(f"[ROUTER] Task: {task}")

        primary = self.routes.get(task, self._gpt_dispatch)

        try:
            return await primary(prompt)
        except Exception as e:
            logger.warning(f"[FALLBACK 1] {e}")

            try:
                return await self._claude_dispatch(prompt)
            except Exception as e:
                logger.warning(f"[FALLBACK 2] {e}")

                try:
                    return await self._xai_dispatch(prompt)
                except Exception as e:
                    logger.error(f"[FAILURE] {e}")
                    raise RuntimeError("All models failed")

    # =========================
    # MODEL DISPATCHERS
    # =========================
    async def _gpt_dispatch(self, prompt: str):
        logger.info("[MODEL] GPT")
        # Integrate OpenAI client here
        return {"model": "gpt", "response": f"GPT response: {prompt}"}

    async def _claude_dispatch(self, prompt: str):
        logger.info("[MODEL] Claude")
        return {"model": "claude", "response": f"Claude response: {prompt}"}

    async def _xai_dispatch(self, prompt: str):
        logger.info("[MODEL] XAI")
        return {"model": "xai", "response": f"XAI response: {prompt}"}

    # =========================
    # TEST ROUTING
    # =========================
    async def test_dispatch(self):
        results = {}
        for key in self.routes:
            results[key] = await self.dispatch(key, "test")
        return results

🔐 3. MERKLE AUDIT CHAIN (TAMPER-PROOF LOGGING)

services/merkle_audit.py

import hashlib
import time
from typing import List

class MerkleAudit:
    def __init__(self):
        self.logs: List[str] = []

    def _hash(self, data: str) -> str:
        return hashlib.sha3_512(data.encode()).hexdigest()

    def add_log(self, event: str):
        timestamp = str(time.time())
        entry = f"{timestamp}:{event}"
        hashed = self._hash(entry)
        self.logs.append(hashed)
        return hashed

    def build_tree(self):
        nodes = self.logs.copy()

        if not nodes:
            return None

        while len(nodes) > 1:
            temp = []
            for i in range(0, len(nodes), 2):
                left = nodes[i]
                right = nodes[i + 1] if i + 1 < len(nodes) else left
                combined = self._hash(left + right)
                temp.append(combined)
            nodes = temp

        return nodes[0]  # Merkle root

    def verify_integrity(self, expected_root: str):
        return self.build_tree() == expected_root

"use client"

import { useState } from "react"

export default function Dashboard() {
  const [prompt, setPrompt] = useState("")
  const [task, setTask] = useState("general")
  const [response, setResponse] = useState("")
  const [loading, setLoading] = useState(false)

  const sendQuery = async () => {
    setLoading(true)

    const res = await fetch("http://localhost:8000/ai/query-stream", {
      method: "POST",
      body: JSON.stringify({ task, prompt }),
    })

    const reader = res.body?.getReader()
    const decoder = new TextDecoder()

    let result = ""

    while (true) {
      const { done, value } = await reader!.read()
      if (done) break

      const chunk = decoder.decode(value)
      result += chunk
      setResponse(result)
    }

    setLoading(false)
  }

  return (
    <main style={{ padding: 20 }}>
      <h1>EDAIL AI Dashboard</h1>

      <select onChange={(e) => setTask(e.target.value)}>
        <option value="general">General (GPT)</option>
        <option value="reasoning">Reasoning (Claude)</option>
        <option value="realtime">Realtime (XAI)</option>
      </select>

      <textarea
        placeholder="Enter prompt..."
        onChange={(e) => setPrompt(e.target.value)}
        style={{ width: "100%", height: 120 }}
      />

      <button onClick={sendQuery} disabled={loading}>
        {loading ? "Streaming..." : "Send"}
      </button>

      <pre>{response}</pre>
    </main>
  )
}

async def query_stream(request: Request):
    body = await request.json()
    task = body["task"]
    prompt = body["prompt"]

    async def generator():
        async for chunk in router_instance.stream(task, prompt):

    return StreamingResponse(generator(), media_type="text/plain")

import hashlib
import os

class PQCrypto:
    def __init__(self):
        self.secret = os.urandom(32)

    # =====================
    # HASHING (BLAKE3 SIM)
    # =====================
    def hash_data(self, data: str):
        return hashlib.blake2b(data.encode()).hexdigest()

    # =====================
    # SIGNATURE (Dilithium-like)
    # =====================
    def sign(self, message: str):
        return self.hash_data(message + str(self.secret))

    def verify(self, message: str, signature: str):
        return self.sign(message) == signature

    # =====================
    # KEY EXCHANGE (ML-KEM SIM)
    # =====================
    def derive_shared_key(self, peer_pub: str):
        return self.hash_data(peer_pub + str(self.secret))

import time, jwt
from passlib.hash import argon2
from services.pq_crypto import PQCrypto

SECRET_KEY = "supersecret_enterprise_key"  # can be PQ-backed

crypto = PQCrypto()

class AuthService:
    def hash_password(self, password: str) -> str:
        return argon2.hash(password)

    def verify_password(self, password: str, hashed: str) -> bool:
        return argon2.verify(password, hashed)

    def create_token(self, user_id: str, role: str) -> str:
        payload = {
            "sub": user_id,
            "role": role,
            "iat": int(time.time()),
            "sig": crypto.sign(user_id + role)
        }
        token = jwt.encode(payload, SECRET_KEY, algorithm="HS256")
        return token

    def verify_token(self, token: str):
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=["HS256"])
            sig_valid = crypto.verify(payload["sub"] + payload["role"], payload["sig"])
            if not sig_valid:
                raise Exception("Invalid signature")
            return payload
        except Exception:
            return None

import logging
from typing import List, Dict, Optional
from .pq_crypto import sign_message
from .merkle_audit import MerkleAudit

logger = logging.getLogger("ModelRouter")
logger.setLevel(logging.INFO)

class ModelRouter:
    def __init__(self, merkle_audit: MerkleAudit):
        # Define primary -> fallback chain
        self.routing: Dict[str, List[str]] = {
            "core": ["gpt", "anthropic", "xai"],
            "healthcare": ["gpt-medical", "claude-health"],
            "defense": ["claude-secure", "xai-defense"],
        }
        self.merkle_audit = merkle_audit

    def dispatch(self, category: str, prompt: str) -> str:
        """
        Try models in fallback order until one succeeds.
        Logs each attempt to Merkle audit chain.
        """
        if category not in self.routing:
            logger.warning(f"No routing for category: {category}")
            return "No available models"

        for model_name in self.routing[category]:
            logger.info(f"Dispatching to {model_name} for category {category}")
            try:
                # Simulate model response
                response = f"{model_name} response to: {prompt}"
                
                # Sign and log to Merkle
                signature = sign_message(response.encode())
                self.merkle_audit.add_entry(model_name, prompt, response, signature)
                
                logger.info(f"Success: {model_name}")
                return response
            except Exception as e:
                logger.error(f"{model_name} failed: {e}")
        logger.error("All fallback models failed")
        return "Failed to generate response"

    def test_routes(self):
        """
        Optional method to test all routing keys
        """
        results = {}
        for cat in self.routing.keys():
            results[cat] = self.dispatch(cat, "Test prompt")
        return results

from cryptography.hazmat.primitives.ciphers.aead import ChaCha20Poly1305
import os
from nacl.signing import SigningKey

# ChaCha20-XChaCha20 encryption
class PQCrypto:
    def __init__(self):
        self.key = ChaCha20Poly1305.generate_key()
        self.signing_key = SigningKey.generate()

    def encrypt(self, plaintext: bytes, nonce: bytes) -> bytes:
        return ChaCha20Poly1305(self.key).encrypt(nonce, plaintext, None)

    def decrypt(self, ciphertext: bytes, nonce: bytes) -> bytes:
        return ChaCha20Poly1305(self.key).decrypt(nonce, ciphertext, None)

def sign_message(message: bytes) -> str:
    key = SigningKey.generate()
    signed = key.sign(message)
    return signed.hex()

        prev_hash = self.chain[-1]['hash'] if self.chain else "0"*64
        entry = {
            "timestamp": datetime.utcnow().isoformat(),
            "model": model,
            "prompt": prompt,
            "response": response,
            "signature": signature,
            "prev_hash": prev_hash,
        }
        entry_hash = self._hash(json.dumps(entry, sort_keys=True))
        entry["hash"] = entry_hash
        self.chain.append(entry)
        return entry_hash

    def verify_chain(self) -> bool:
        for i in range(1, len(self.chain)):
            prev = self.chain[i-1]["hash"]
            if self.chain[i]["prev_hash"] != prev:
                return False
        return True

import crypto from "crypto";
import { auditService } from "../services/auditService";
import { GPTClient } from "./workers/gptClient";
import { AnthropicClient } from "./workers/anthropicClient";
import { XAIClient } from "./workers/xaiClient";

type ModelName = "GPT" | "Anthropic" | "XAI";

export class ModelRouter {
  private clients: Record<ModelName, any>;
  private fallbackChain: ModelName[];

  constructor() {
    this.clients = {
      GPT: new GPTClient(),
      Anthropic: new AnthropicClient(),
      XAI: new XAIClient(),
    };
    // Priority fallback chain
    this.fallbackChain = ["GPT", "Anthropic", "XAI"];
  }

  async dispatch(prompt: string, context?: any): Promise<any> {
    for (const model of this.fallbackChain) {
      try {
        const response = await this.clients[model].generate(prompt, context);
        // Log response to Merkle audit
        await auditService.log(model, prompt, response);
        return response;
      } catch (err) {
        console.error(`[ModelRouter] ${model} failed:`, err);
      }
    }
    throw new Error("All AI models failed.");
  }
}

2. backend/src/services/auditService.ts (Merkle Chain)

import crypto from "crypto";

interface AuditLog {
  timestamp: number;
  model: string;
  promptHash: string;
  responseHash: string;
  prevHash?: string;
}

class MerkleAuditService {
  private chain: AuditLog[] = [];

  async log(model: string, prompt: string, response: string) {
    const prevHash = this.chain.length > 0 ? this.chain[this.chain.length - 1].responseHash : "";
    const entry: AuditLog = {
      timestamp: Date.now(),
      model,
      promptHash: crypto.createHash("sha3-512").update(prompt).digest("hex"),
      responseHash: crypto.createHash("sha3-512").update(response + prevHash).digest("hex"),
      prevHash,
    };
    this.chain.push(entry);
    console.log(`[AuditService] Logged response for ${model} at ${entry.timestamp}`);
  }

  getChain() {
    return this.chain;
  }
}

export const auditService = new MerkleAuditService();

const prisma = new PrismaClient();
const SECRET = process.env.JWT_SECRET || "supersecret";

export async function login(email: string, password: string) {
  const user = await prisma.user.findUnique({ where: { email } });
  if (!user) throw new Error("User not found");
  const valid = await bcrypt.compare(password, user.password);
  if (!valid) throw new Error("Invalid credentials");

  const token = jwt.sign({ userId: user.id, role: user.role }, SECRET, { expiresIn: "2h" });
  return { token, user };
}

merkleAudit.ts

import { createHash } from "crypto";

export class MerkleTree {
  leaves: string[];
  layers: string[][] = [];

  constructor(leaves: string[]) {
    this.leaves = leaves.map(l => this.hash(l));
    this.buildTree();
  }

  private hash(data: string) {
    return createHash("sha256").update(data).digest("hex");
  }

  private buildTree() {
    this.layers = [this.leaves];
    let currentLayer = this.leaves;
    while (currentLayer.length > 1) {
      const nextLayer = [];
      for (let i = 0; i < currentLayer.length; i += 2) {
        if (i + 1 === currentLayer.length) nextLayer.push(currentLayer[i]);
        else nextLayer.push(this.hash(currentLayer[i] + currentLayer[i + 1]));
      }
      currentLayer = nextLayer;
      this.layers.unshift(currentLayer);
    }
  }

  getRoot() {
    return this.layers[0][0];
  }
}

import crypto from "crypto";
import { PrismaClient } from "@prisma/client";
import { logger } from "./logger";

const prisma = new PrismaClient();

export class MerkleAudit {
  private chain: string[] = [];

  constructor() {}

  hash(data: string): string {
    return crypto.createHash("sha256").update(data).digest("hex");
  }

  addLog(action: string, payload: any) {
    const payloadStr = JSON.stringify(payload);
    const prevHash = this.chain.length > 0 ? this.chain[this.chain.length - 1] : "";
    const currentHash = this.hash(prevHash + payloadStr + action + Date.now());
    this.chain.push(currentHash);

    prisma.auditLog.create({
      data: {
        action,
        payload,
        hash: currentHash
      }
    }).then(() => logger.info(`Audit logged: ${action}`))
      .catch(err => logger.error(`Audit failed: ${err}`));

    return currentHash;
  }

  getRoot(): string {
    return this.chain.length > 0 ? this.chain[this.chain.length - 1] : "";
  }
}

3️⃣ Post-Quantum Crypto Layer (backend/src/pqCrypto.ts)

import * as crypto from "crypto";
import { PrismaClient } from "@prisma/client";
const prisma = new PrismaClient();

export class PQCrypto {
  // Generate ephemeral PQ key (for demo, using x25519 as placeholder)
  static async generateKey() {
    const { publicKey, privateKey } = crypto.generateKeyPairSync("x25519");
    const pub = publicKey.export({ type: "spki", format: "pem" }).toString();
    const priv = privateKey.export({ type: "pkcs8", format: "pem" }).toString();

    const now = new Date();
    const validTo = new Date(now.getTime() + 90 * 24 * 60 * 60 * 1000); // 90 days

    await prisma.postQuantumKey.create({
      data: {
        publicKey: pub,
        privateKey: priv,
        validFrom: now,
        validTo
      }
    });

    return { pub, priv };
  }

  static encrypt(data: string, pubKey: string): string {
    const buffer = Buffer.from(data);
    // For demo: XOR-based placeholder
    const keyBuffer = Buffer.from(pubKey.slice(0, 32));
    const encrypted = buffer.map((b, i) => b ^ keyBuffer[i % keyBuffer.length]);
    return encrypted.toString("base64");
  }

  static decrypt(encData: string, privKey: string): string {
    const buffer = Buffer.from(encData, "base64");
    const keyBuffer = Buffer.from(privKey.slice(0, 32));
    const decrypted = buffer.map((b, i) => b ^ keyBuffer[i % keyBuffer.length]);
    return Buffer.from(decrypted).toString();
  }
}

  }

  async dispatch(modelName: string, payload: any): Promise<any> {
    const tried: string[] = [];
    const attemptChain = async (name: string): Promise<any> => {
      if (!this.models[name]) {
        logger.warn(`Model not found: ${name}`);
        throw new Error("Model not found: " + name);
      }

      const model = this.models[name];
      tried.push(name);

      try {
        const response = await fetch(model.url, {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            ...(model.apiKey ? { "Authorization": `Bearer ${model.apiKey}` } : {})
          },
          body: JSON.stringify(payload)
        });

        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const result = await response.json();
        logger.info(`Model ${name} response OK`);
        return result;
      } catch (err) {
        logger.error(`Model ${name} failed: ${err}`);
        if (model.fallback && model.fallback.length > 0) {
          for (const fallbackName of model.fallback) {
            if (!tried.includes(fallbackName)) {
              return attemptChain(fallbackName);
            }
          }
        }
        throw err;
      }
    };

    return attemptChain(modelName);
  }
}

import { useEffect, useRef, useState } from "react";

export default function useWebSocket(url: string) {
  const ws = useRef<WebSocket | null>(null);
  const [messages, setMessages] = useState<{ user: string; text: string }[]>([]);

  useEffect(() => {
    ws.current = new WebSocket(url);
    ws.current.onmessage = (event) => {
      const msg = JSON.parse(event.data);
      setMessages((prev) => [...prev, msg]);
    };
    return () => ws.current?.close();
  }, [url]);

  const sendMessage = (text: string) => {
    ws.current?.send(JSON.stringify({ user: "frontend", text }));
  };

  return { messages, sendMessage };
}

2️⃣ Merkle Audit Chain

import crypto from "crypto";

interface AuditEntry {
  timestamp: number;
  prompt: string;
  response: string;
  model: string;
  hash: string;
  prevHash: string;
}

let auditChain: AuditEntry[] = [];

export async function logAudit(prompt: string, response: string, model: string) {
  const prevHash = auditChain.length ? auditChain[auditChain.length - 1].hash : "0".repeat(64);
  const hash = crypto.createHash("sha256").update(prompt + response + model + prevHash).digest("hex");
  
  auditChain.push({ timestamp: Date.now(), prompt, response, model, hash, prevHash });
  return hash;
}

export function verifyChain() {
  for (let i = 1; i < auditChain.length; i++) {
    const { hash, prevHash, prompt, response, model } = auditChain[i];
    const calc = crypto.createHash("sha256").update(prompt + response + model + auditChain[i-1].hash).digest("hex");
    if (hash !== calc) throw new Error(`Audit chain compromised at index ${i}`);
  }
  return true;
}

import crypto from "crypto";

export function generateKeyPair() {
  const { publicKey, privateKey } = crypto.generateKeyPairSync("rsa", { modulusLength: 4096 });

export function encryptMessage(msg: string, pub: crypto.KeyObject) {
  return crypto.publicEncrypt(pub, Buffer.from(msg)).toString("base64");
}

export function decryptMessage(enc: string, priv: crypto.KeyObject) {
  return crypto.privateDecrypt(priv, Buffer.from(enc, "base64")).toString();

1️⃣ Backend: ModelRouter + Merkle Audit Chain

File: backend/src/services/modelRouter.ts

import { MerkleChain } from './merkleChain';
import { getModelResponse } from './modelClients';
import logger from '../utils/logger';

interface ModelOptions {
  name: string;
  apiKey: string;
  enabled: boolean;
}

export class ModelRouter {
  private models: ModelOptions[];
  private merkle: MerkleChain;

  constructor(models: ModelOptions[]) {
    this.models = models;
    this.merkle = new MerkleChain();
  }

  async dispatch(prompt: string): Promise<string> {
    let lastError = null;

    for (const model of this.models) {
      if (!model.enabled) continue;
      try {
        const response = await getModelResponse(model.name, prompt, model.apiKey);
        this.merkle.addRecord(model.name, prompt, response);
        return response;
      } catch (err) {
        lastError = err;
        logger.error(`Model ${model.name} failed: ${err}`);
      }
    }

    this.merkle.addRecord('fallback_failed', prompt, lastError?.toString() || 'Unknown error');
    throw lastError || new Error('All models failed.');
  }

  getAuditRoot() {
    return this.merkle.getRoot();
  }
}

File: backend/src/services/merkleChain.ts

import crypto from 'crypto';
import { sha3_512 } from 'js-sha3';

interface RecordEntry {
  model: string;
  prompt: string;
  response: string;
  hash: string;
}

export class MerkleChain {
  private records: RecordEntry[] = [];
  private root: string = '';

  addRecord(model: string, prompt: string, response: string) {
    const hash = sha3_512(`${model}:${prompt}:${response}`);
    this.records.push({ model, prompt, response, hash });
    this.root = this.computeRoot();
  }

  private computeRoot(): string {
    let hashes = this.records.map(r => r.hash);
    while (hashes.length > 1) {
      const temp: string[] = [];
      for (let i = 0; i < hashes.length; i += 2) {
        if (i + 1 < hashes.length) {
          temp.push(sha3_512(hashes[i] + hashes[i + 1]));
        } else {
          temp.push(hashes[i]);
        }
      }
      hashes = temp;
    }
    return hashes[0] || '';
  }

  getRoot() {
    return this.root;
  }

  getRecords() {
    return this.records;
  }
}

import crypto from 'crypto';

// Example: PQ-secure key derivation & encryption (simplified)
export function encryptPQ(message: string, key: Buffer) {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv('chacha20-poly1305', key, iv, { authTagLength: 16 });
  let encrypted = cipher.update(message, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  const tag = cipher.getAuthTag().toString('hex');
  return { encrypted, iv: iv.toString('hex'), tag };
}

Merkle Audit Chain

// backend/src/services/merkleAudit.ts
import crypto from 'crypto';

export class MerkleTree {
  leaves: string[];
  constructor() {
    this.leaves = [];
  }

  addLeaf(data: string) {
    const hash = crypto.createHash('sha256').update(data).digest('hex');
    this.leaves.push(hash);
    return hash;
  }

  getRoot(): string {
    let nodes = [...this.leaves];
    while (nodes.length > 1) {
      const temp: string[] = [];
      for (let i = 0; i < nodes.length; i += 2) {
        const left = nodes[i];
        const right = nodes[i + 1] || left;
        const hash = crypto.createHash('sha256').update(left + right).digest('hex');
        temp.push(hash);
      }
      nodes = temp;
    }
    return nodes[0];
  }
}

merkleAudit.ts

import crypto from "crypto";

export class MerkleNode {
  hash: string;
  left?: MerkleNode;
  right?: MerkleNode;

  constructor(left: MerkleNode | null, right: MerkleNode | null, data?: string) {
    if (!left && !right && data) {
      this.hash = crypto.createHash("sha256").update(data).digest("hex");
    } else {
      this.left = left!;
      this.right = right!;
      this.hash = crypto.createHash("sha256")
        .update(left!.hash + right!.hash)
        .digest("hex");
    }
  }
}

export class MerkleTree {
  root: MerkleNode;
  constructor(leaves: string[]) {
    let nodes = leaves.map(d => new MerkleNode(null, null, d));
    while (nodes.length > 1) {
      const newLevel: MerkleNode[] = [];
      for (let i = 0; i < nodes.length; i += 2) {
        if (i + 1 < nodes.length) {
          newLevel.push(new MerkleNode(nodes[i], nodes[i + 1]));
        } else {
          newLevel.push(nodes[i]);
        }
      }
      nodes = newLevel;
    }
    this.root = nodes[0];
  }

  getRootHash() {
    return this.root.hash;
  }
}

import crypto from "crypto";
import fs from "fs";
import path from "path";

interface LogEntry {
  timestamp: string;
  action: string;
  user: string;
  details: string;
}

export class MerkleAudit {
  private entries: LogEntry[] = [];
  private merkleRoot: string = "";

  addEntry(entry: LogEntry) {
    this.entries.push(entry);
    this.updateMerkleRoot();
  }

  private updateMerkleRoot() {
    const hashes = this.entries.map((e) =>
      crypto.createHash("sha256").update(JSON.stringify(e)).digest("hex")
    );
    this.merkleRoot = this.buildMerkleRoot(hashes);
    fs.writeFileSync(
      path.join(__dirname, "../../logs/merkleRoot.log"),
      this.merkleRoot
    );
  }

  private buildMerkleRoot(hashes: string[]): string {
    if (hashes.length === 1) return hashes[0];
    const nextLevel: string[] = [];
    for (let i = 0; i < hashes.length; i += 2) {
      const left = hashes[i];
      const right = i + 1 < hashes.length ? hashes[i + 1] : left;
      const combined = crypto.createHash("sha256").update(left + right).digest("hex");
      nextLevel.push(combined);
    }
    return this.buildMerkleRoot(nextLevel);
  }

  getRoot(): string {
    return this.merkleRoot;
  }
}

3.1 Post-Quantum Crypto Service

pq-encryption.service.ts

import { randomBytes, createCipheriv, createDecipheriv } from "crypto";
import { generateKeyPair } from "liboqs-node"; // NIST PQC candidate lib

export class PQCryptoService {
  private keyPair: { publicKey: Buffer; privateKey: Buffer };

  constructor() {
    this.keyPair = generateKeyPair("Dilithium"); // Post-Quantum Signature
  }

  encrypt(data: string): string {
    const iv = randomBytes(12);
    const cipher = createCipheriv("chacha20-poly1305", this.keyPair.publicKey.slice(0, 32), iv, { authTagLength: 16 });
    const encrypted = Buffer.concat([cipher.update(data, "utf8"), cipher.final()]);
    const tag = cipher.getAuthTag();
    return Buffer.concat([iv, tag, encrypted]).toString("base64");
  }

  decrypt(encryptedData: string): string {
    const raw = Buffer.from(encryptedData, "base64");
    const iv = raw.slice(0, 12);
    const tag = raw.slice(12, 28);
    const encrypted = raw.slice(28);
    const decipher = createDecipheriv("chacha20-poly1305", this.keyPair.privateKey.slice(0, 32), iv, { authTagLength: 16 });
    decipher.setAuthTag(tag);
    return decipher.update(encrypted) + decipher.final("utf8");
  }

  sign(data: string): string {
    return this.keyPair.privateKey.toString("hex"); // Simplified, use real Dilithium signing
  }

  getPublicKey(): string {
    return this.keyPair.publicKey.toString("hex");
  }
}

backend/src/utils/merkle.util.ts

import crypto from 'crypto';

export class MerkleUtil {
  private leaves: string[] = [];
  private root: string | null = null;

  addLeaf(dataHash: string) {
    this.leaves.push(dataHash);
    this.root = this.computeRoot();
  }

  computeRoot(): string | null {
    if (!this.leaves.length) return null;
    let nodes = [...this.leaves];
    while (nodes.length > 1) {
      const temp: string[] = [];
      for (let i = 0; i < nodes.length; i += 2) {
        const left = nodes[i];
        const right = nodes[i + 1] || left;
        temp.push(
          crypto.createHash('sha256').update(left + right).digest('hex'),
        );
      }
      nodes = temp;
    }
    return nodes[0];
  }

  getRoot() {
    return this.root;
  }

  getLeaves() {
    return this.leaves;
  }
}

import * as sodium from 'libsodium-wrappers';

export async function generateKeyPair() {
  await sodium.ready;
  const keyPair = sodium.crypto_sign_keypair();
  return {
    publicKey: sodium.to_hex(keyPair.publicKey),
    privateKey: sodium.to_hex(keyPair.privateKey),
  };
}

export async function signMessage(privateKeyHex: string, message: string) {
  await sodium.ready;
  const privateKey = sodium.from_hex(privateKeyHex);
  const signed = sodium.crypto_sign_detached(
    sodium.from_string(message),
    privateKey
  );
  return sodium.to_hex(signed);
}

export async function verifySignature(publicKeyHex: string, message: string, signatureHex: string) {
  await sodium.ready;
  const publicKey = sodium.from_hex(publicKeyHex);
  const signature = sodium.from_hex(signatureHex);
  return sodium.crypto_sign_verify_detached(signature, sodium.from_string(message), publicKey);
}
