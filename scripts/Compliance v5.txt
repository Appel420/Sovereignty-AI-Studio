# agip_core/policy.py

from dataclasses import dataclass, replace
from hashlib import sha256
import json


class PolicyValidationError(ValueError):
    pass


@dataclass(frozen=True)
class MetricPublicationPolicy:
    requires_snapshot: bool = True
    requires_verified_evidence: bool = True
    requires_control_mapping: bool = True
    requires_framework_alignment: bool = True
    mutable_registry_access: bool = False
    synthetic_metrics: bool = False

    policy_version: str = "v1"
    policy_hash: str = ""


def canonical_policy_payload(
    policy: MetricPublicationPolicy,
) -> dict:
    return {
        "requires_snapshot": policy.requires_snapshot,
        "requires_verified_evidence": policy.requires_verified_evidence,
        "requires_control_mapping": policy.requires_control_mapping,
        "requires_framework_alignment": policy.requires_framework_alignment,
        "mutable_registry_access": policy.mutable_registry_access,
        "synthetic_metrics": policy.synthetic_metrics,
        "policy_version": policy.policy_version,
    }


def compute_policy_hash(
    policy: MetricPublicationPolicy,
) -> str:
    payload = json.dumps(
        canonical_policy_payload(policy),
        sort_keys=True,
        separators=(",", ":"),
    )

    return sha256(
        payload.encode("utf-8")
    ).hexdigest()


def validate_metric_policy(
    policy: MetricPublicationPolicy,
) -> MetricPublicationPolicy:

    if not policy.requires_snapshot:
        raise PolicyValidationError(
            "Snapshot requirement cannot be disabled"
        )

    if not policy.requires_verified_evidence:
        raise PolicyValidationError(
            "Verified evidence requirement cannot be disabled"
        )

    if not policy.requires_control_mapping:
        raise PolicyValidationError(
            "Control mapping requirement cannot be disabled"
        )

    if not policy.requires_framework_alignment:
        raise PolicyValidationError(
            "Framework alignment requirement cannot be disabled"
        )

    if policy.mutable_registry_access:
        raise PolicyValidationError(
            "Mutable registry access forbidden"
        )

    if policy.synthetic_metrics:
        raise PolicyValidationError(
            "Synthetic metrics forbidden"
        )

    computed_hash = compute_policy_hash(policy)

    if policy.policy_hash and policy.policy_hash != computed_hash:
        raise PolicyValidationError(
            "Policy hash mismatch"
        )

    return replace(
        policy,
        policy_hash=computed_hash,
    )

    requires_control_mapping: bool = True
    requires_framework_alignment: bool = True
    mutable_registry_access: bool = False
    synthetic_metrics: bool = False

    policy_hash: str = ""

def canonical_policy_payload(policy: MetricPublicationPolicy) -> dict:
    return {
        "requires_snapshot": policy.requires_snapshot,
        "requires_verified_evidence": policy.requires_verified_evidence,
        "requires_control_mapping": policy.requires_control_mapping,
        "requires_framework_alignment": policy.requires_framework_alignment,
        "mutable_registry_access": policy.mutable_registry_access,
        "synthetic_metrics": policy.synthetic_metrics,
        "policy_version": policy.policy_version,
    }

def compute_policy_hash(policy: MetricPublicationPolicy) -> str:
    payload = json.dumps(
        canonical_policy_payload(policy),
        sort_keys=True,
        separators=(",", ":"),
    )
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()

def validate_metric_policy(
    policy: MetricPublicationPolicy,
) -> MetricPublicationPolicy:
    required_true = [
        "requires_snapshot",
        "requires_verified_evidence",
        "requires_control_mapping",
        "requires_framework_alignment",
    ]
    for field_name in required_true:
        if getattr(policy, field_name) is not True:
            raise PolicyValidationError(
                f"Policy relaxation rejected: {field_name}"
            )
    if policy.mutable_registry_access:
        raise PolicyValidationError(
            "Mutable registry access is forbidden"
        )
    if policy.synthetic_metrics:
        raise PolicyValidationError(
            "Synthetic metrics are forbidden"
        )
    computed_hash = compute_policy_hash(policy)
    if policy.policy_hash and policy.policy_hash != computed_hash:
        raise PolicyValidationError(
            "Policy hash mismatch"
        )
    return replace(
        policy,
        policy_hash=computed_hash,
    )

    policy: MetricPublicationPolicy,
) -> tuple[MetricPublicationPolicy, PolicyAdmissionEvent]:

    event = PolicyAdmissionEvent(
        policy_version=validated.policy_version,
        policy_hash=validated.policy_hash,
        admitted_at=datetime.now(timezone.utc).isoformat(),

    return validated, event
