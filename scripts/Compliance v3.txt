```python
#!/usr/bin/env python3
import json
import sys
from pathlib import Path

def load_policy(path: Path) -> dict:
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)

def validate_policy(policy: dict) -> None:
    required = ["policy_id", "enforcement", "protected_paths", "forbidden_patterns"]
    for key in required:
        if key not in policy:
            raise ValueError(f"Missing policy field: {key}")

    enforcement = policy["enforcement"]
    if enforcement.get("mode") != "deny":
        raise ValueError("Security policy must use deny mode")
    if enforcement.get("security_modules_strict") is not True:
        raise ValueError("Security policy must enforce strict security modules")
    if not policy["protected_paths"]:
        raise ValueError("Security policy must define protected paths")
    if not policy["forbidden_patterns"]:
        raise ValueError("Security policy must define forbidden patterns")

def is_governed_file(path: Path, root: Path, protected_paths: list[str]) -> bool:
    try:
        resolved = path.resolve()
        root_resolved = root.resolve()
    except OSError:
        return False

    if root_resolved not in resolved.parents and resolved != root_resolved:
        return False

    rel = resolved.relative_to(root_resolved)
    rel_text = rel.as_posix()
    return any(rel_text.startswith(p.rstrip("/")) for p in protected_paths)

def scan_file(path: Path, root: Path, protected_paths: list[str], patterns: list[str]) -> list[str]:
    if not is_governed_file(path, root, protected_paths):
        return []

    findings = []
    try:
        content = path.read_text(encoding="utf-8", errors="ignore")
    except OSError:
        return findings

    for pattern in patterns:
        if pattern in content:
            findings.append(pattern)
    return findings

def collect_targets(root: Path, protected_paths: list[str]):
    targets = []
    for protected in protected_paths:
        directory = root / protected
        if not directory.exists():
            continue
        for file in directory.rglob("*"):
            if file.is_file():
                targets.append(file)
    return targets

def main() -> int:
    if len(sys.argv) != 3:
        print("usage: security_policy_scan.py POLICY ROOT")
        return 1

    policy_path = Path(sys.argv[1])
    root = Path(sys.argv[2])

    policy = load_policy(policy_path)
    validate_policy(policy)

    patterns = []
    for entry in policy["forbidden_patterns"]:
        patterns.extend(entry["patterns"])

    findings = []
    for file in collect_targets(root, policy["protected_paths"]):
        matches = scan_file(file, root, policy["protected_paths"], patterns)
        for match in matches:
            findings.append({"file": str(file), "pattern": match})

    if findings:
        print("ZERO_TOLERANCE_VIOLATIONS")
        for finding in findings:
            print(f'{finding["file"]}: {finding["pattern"]}')
        return 1

    print("ZERO_TOLERANCE_OK")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

    payload = json.dumps(

import json
from hashlib import sha512
from blake3 import blake3

def normalize_findings(findings: list[dict]) -> list[dict]:
    return sorted(
        findings,
        key=lambda item: (
            item.get("file", ""),
            item.get("pattern", ""),
            item.get("category", ""),
        ),
    )

def create_scan_id(policy_id: str, repository: str, findings: list[dict]) -> str:
    normalized = normalize_findings(findings)
    payload = json.dumps(
        {
            "policy_id": policy_id,
            "repository": repository,
            "findings": normalized,
        },
        sort_keys=True,
        separators=(",", ":"),
    ).encode("utf-8")
    return blake3(payload).hexdigest()

def create_scan_fingerprint(policy_id: str, repository: str, findings: list[dict]) -> str:

            "repository": repository,
            "findings": normalized,
        },
        sort_keys=True,
        separators=(",", ":"),

from sovereignty_playground.ledger import SovereigntyLedger

def test_full_stack_workflow(tmp_path: Path):

    root = RootAuthority(tmp_path / "device.key")
    ledger = SovereigntyLedger(tmp_path / "ledger_state", root)

    assert policy.check("xai-grok").allowed
    assert policy.check("google-gemini").allowed is False
    assert policy.check("some-unreviewed-thing").allowed is False  # fail-closed

        actor="derek",
    )

    assert identity.check(token_id="tok-audit-1", permission="tool.shell_exec") is False

This is the right direction for the AGIP domain layer, but I would tighten the contracts to preserve the invariant you established:
No canonical object exists without verifiable evidence.
The current dataclasses allow invalid states:
	•	Model.evidence_refs can be empty.
	•	EvidenceRef.verified can be True without proving verification source/state.
	•	DashboardMetric can hold an unverified metric.
	•	value: str prevents reliable metric lineage and calculations.
A stronger v1 domain model:
from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from typing import List


class VerificationStatus(str, Enum):
    PENDING = "pending"
    VERIFIED = "verified"
    REJECTED = "rejected"


@dataclass(frozen=True)
class EvidenceRef:
    evidence_id: str
    source: str
    timestamp: datetime
    verification_status: VerificationStatus
    confidence: float

    def __post_init__(self):
        if not self.evidence_id.strip():
            raise ValueError("evidence_id required")

        if not 0.0 <= self.confidence <= 1.0:
            raise ValueError("confidence must be between 0 and 1")

        if self.verification_status != VerificationStatus.VERIFIED:
            raise ValueError(
                "Only verified evidence may be referenced by canonical objects"
            )


@dataclass
class Model:
    id: str
    name: str
    provider_id: str
    lifecycle: str
    evidence_refs: List[EvidenceRef] = field(default_factory=list)

    def __post_init__(self):
        if not self.evidence_refs:
            raise ValueError(
                "Model registration requires evidence references"
            )


@dataclass(frozen=True)
class DashboardMetric:
    key: str
    value: float
    evidence_refs: List[EvidenceRef]
    verification_status: VerificationStatus

    def __post_init__(self):
        if not self.evidence_refs:
            raise ValueError(
                "Metrics require evidence lineage"
            )

        if self.verification_status != VerificationStatus.VERIFIED:
            raise ValueError(
                "Only verified metrics may be published"
            )

        for evidence in self.evidence_refs:
            if evidence.verification_status != VerificationStatus.VERIFIED:
                raise ValueError(
                    "Metric contains unverified evidence"
                )

from agip_core.domain.model import Model
from agip_core.domain.metric import DashboardMetric
from agip_core.domain.evidence import VerificationStatus


class AdmissionError(ValueError):
    pass


def admit_model(model: Model) -> Model:
    if not model.evidence_refs:
        raise AdmissionError(
            "Canonical model requires at least one evidence reference"
        )

    for evidence in model.evidence_refs:
        if not evidence.verified:
            raise AdmissionError(
                f"Unverified evidence rejected: {evidence.evidence_id}"
            )

    return model


def publish_metric(metric: DashboardMetric) -> DashboardMetric:
    if metric.verification_status != VerificationStatus.VERIFIED:

        raise AdmissionError(

    if not metric.evidence_ref.verified:

import pytest

from agip_core.admission import (
    admit_model,
    publish_metric,
    AdmissionError,

from agip_core.domain.metric import DashboardMetric

    EvidenceRef,
    VerificationStatus,

def verified_evidence():
    return EvidenceRef(
        evidence_id="ev_001",
        source="provider_release",
        verified=True,
    )


def unverified_evidence():
    return EvidenceRef(
        evidence_id="ev_002",
        source="unknown",
        verified=False,
    )


def test_model_requires_evidence():
    model = Model(
        id="model_001",
        name="Example",
        provider_id="provider_001",
        lifecycle="active",
        evidence_refs=[],
    )

        admit_model(model)


def test_model_rejects_unverified_evidence():
    model = Model(
        id="model_001",
        name="Example",
        provider_id="provider_001",
        lifecycle="active",
        evidence_refs=[
            unverified_evidence()
        ],
    )

    with pytest.raises(AdmissionError):
        admit_model(model)


def test_model_accepts_verified_evidence():
    model = Model(
        id="model_001",
        name="Example",
        provider_id="provider_001",
        lifecycle="active",
        evidence_refs=[
            verified_evidence()
        ],
