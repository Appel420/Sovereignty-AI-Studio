class ModelRegistry:
    def __init__(self):
        self.models = {}
        self.lock = threading.Lock()

    def register_model(self, model_name, provider, policy):
        """
        Register a new model in the registry.
        """
        with self.lock:
            if model_name in self.models:
                print(f"Model '{model_name}' already exists.")
            else:
                self.models[model_name] = {
                    'provider': provider,
                    'policy': policy
                }
                print(f"Model '{model_name}' registered successfully.")

    def get_model(self, model_name):
        """
        Retrieve details of a registered model.
        """
        with self.lock:
            if model_name in self.models:
                model = self.models[model_name]
                print(f"Model '{model_name}' details:")
                print(f"  Provider: {model['provider']}")
                print(f"  Policy: {model['policy']}")
            else:
                print(f"Model '{model_name}' not found.")

    def list_models(self):
        """
        List all registered models.
        """
        with self.lock:
            if self.models:
                print("Registered Models:")
                for model_name, details in self.models.items():
                    print(f"  - {model_name}: Provider={details['provider']}, Policy={details['policy']}")
            else:
                print("No models registered yet.")

    def delete_model(self, model_name):
        """
        Delete a model from the registry.
        """
        with self.lock:
            if model_name in self.models:
                del self.models[model_name]
                print(f"Model '{model_name}' deleted successfully.")
            else:
                print(f"Model '{model_name}' not found.")

# Example usage
registry = ModelRegistry()
registry.register_model("Grok-4.5 fallback -->4.3", "xai", "default")
registry.register_model("gpt-5.6-Luna fallback --> Codex-5.5", "openai", "strict")
registry.register_model("Claude-sonnet-5 fallback--> 4.7", "anthropic", "review")
registry.list_models()
registry.get_model("Grok-4.5 fallback -->4.3")
registry.delete_model("Grok-4.5 fallback -->4.3")
registry.list_models()
