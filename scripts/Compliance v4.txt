from agip_core.domain.evidence import Evidence, VerificationStatus
from agip_core.domain.model import Model


class AdmissionError(ValueError):
    pass


def admit_model(
    model: Model,
    evidence_store: dict[str, Evidence],
) -> Model:

    if not model.id.strip():
        raise AdmissionError("Model requires ID")

    if not model.evidence_refs:
        raise AdmissionError(
            "Canonical model requires evidence lineage"
        )

    for evidence_id in model.evidence_refs:

        if evidence_id not in evidence_store:
            raise AdmissionError(
                f"Missing evidence: {evidence_id}"
            )

        evidence = evidence_store[evidence_id]

        evidence.validate()

        if evidence.object_type != "model":
            raise AdmissionError(
                "Evidence object type mismatch"
            )

        if evidence.verification_status != VerificationStatus.VERIFIED:
            raise AdmissionError(
                "Unverified evidence cannot admit model"
            )

    return model

```python
from agip_core.domain.events import AdmissionEvent


class RegistryError(ValueError):
    pass


@dataclass
class Registry:
    events: list[AdmissionEvent] = field(default_factory=list)

    def append_event(self, event: AdmissionEvent) -> None:
        if event.action not in {"admit", "reject"}:
            raise RegistryError("Invalid event action")
        self.events.append(event)

    def current_state(self) -> dict[str, dict[str, Any]]:
        state: dict[str, dict[str, Any]] = {}
        for event in self.events:
            if event.action == "admit":
                state[event.entity_id] = {
                    "id": event.entity_id,
                    "entity_type": event.entity_type,
                    "evidence_ids": list(event.evidence_ids),
                    "admitted_at": event.timestamp,
                }
            elif event.action == "reject":
                state.pop(event.entity_id, None)
        return state
```

```python
# agip_core/snapshots.py
from dataclasses import dataclass
from hashlib import sha256
import json
from typing import Any


@dataclass(frozen=True)
class Snapshot:
    snapshot_id: str
    parent_hash: str
    state_hash: str
    event_count: int
    payload: dict[str, Any]


def canonical_json(obj: Any) -> str:
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), default=str)


def compute_hash(payload: Any) -> str:
    return sha256(canonical_json(payload).encode("utf-8")).hexdigest()


def make_snapshot(state: dict[str, Any], parent_hash: str = "") -> Snapshot:
    state_hash = compute_hash(state)
    payload = {
        "parent_hash": parent_hash,
        "state_hash": state_hash,
        "state": state,
    }
    snapshot_id = compute_hash(payload)
    return Snapshot(
        snapshot_id=snapshot_id,
        parent_hash=parent_hash,
        state_hash=state_hash,
        event_count=len(state),
        payload=payload,
    )
```

    AdmissionEvent,

class RegistryError(ValueError):

@dataclass(frozen=True)
class SealedEvent:
    event: AdmissionEvent
    previous_hash: str
    event_hash: str


@dataclass
class Registry:
    events: list[SealedEvent] = field(default_factory=list)

    def append_event(self, event: AdmissionEvent) -> None:
        if event.action not in {"admit", "reject"}:
            raise RegistryError("Invalid event action")

        previous_hash = (
            self.events[-1].event_hash
            if self.events
            else ""
        )

        event_hash = hash_event(event, previous_hash)

        self.events.append(
            SealedEvent(
                event=event,
                previous_hash=previous_hash,
                event_hash=event_hash,
            )
        )

    def verify_chain(self) -> bool:
        previous_hash = ""

        for sealed in self.events:
            expected = hash_event(
                sealed.event,
                previous_hash,
            )

            if expected != sealed.event_hash:
                return False

            if sealed.previous_hash != previous_hash:
                return False

            previous_hash = sealed.event_hash

        return True

from dataclasses import dataclass
from hashlib import sha256
import json


class MetricPolicyError(ValueError):
    pass


@dataclass(frozen=True)
class MetricPublicationPolicy:
    requires_snapshot: bool = True
    requires_verified_evidence: bool = True
    requires_control_mapping: bool = True
    requires_framework_alignment: bool = True
    mutable_registry_access: bool = False
    synthetic_metrics: bool = False

    policy_version: str = "v1"
    policy_hash: str = ""

    def canonical_payload(self) -> dict:
        return {
            "requires_snapshot": self.requires_snapshot,
            "requires_verified_evidence": self.requires_verified_evidence,
            "requires_control_mapping": self.requires_control_mapping,
            "requires_framework_alignment": self.requires_framework_alignment,
            "mutable_registry_access": self.mutable_registry_access,
            "synthetic_metrics": self.synthetic_metrics,
            "policy_version": self.policy_version,
        }

    def calculated_hash(self) -> str:
        payload = json.dumps(
            self.canonical_payload(),
            sort_keys=True,
            separators=(",", ":"),
        )

        return sha256(payload.encode("utf-8")).hexdigest()


def validate_metric_policy(
    policy: MetricPublicationPolicy,
) -> MetricPublicationPolicy:

    required_true = [
        "requires_snapshot",
        "requires_verified_evidence",
        "requires_control_mapping",
        "requires_framework_alignment",
    ]

    for field_name in required_true:
        if getattr(policy, field_name) is not True:
            raise MetricPolicyError(
                f"Policy relaxation rejected: {field_name}"
            )

    if policy.mutable_registry_access:
        raise MetricPolicyError(
            "Mutable registry access is forbidden"
        )

    if policy.synthetic_metrics:
        raise MetricPolicyError(
            "Synthetic metrics are forbidden"
        )

    calculated = policy.calculated_hash()

    if policy.policy_hash and policy.policy_hash != calculated:
        raise MetricPolicyError(
            "Policy hash mismatch"
        )

    return MetricPublicationPolicy(
        **policy.canonical_payload(),
        policy_hash=calculated,
    )
