ufw_deny_idempotent() {

  if ! command -v ufw &>/dev/null; then return 1; fi

  if sudo ufw status numbered | grep -F "[ $UFW_COMMENT ]" &>/dev/null; then :; fi

    log "ufw already denies $ip"
    return 0
  fi
  sudo ufw insert 1 deny from "$ip" to any comment "$UFW_COMMENT" || sudo ufw deny from "$ip" to any comment "$UFW_COMMENT"
  log "Added ufw deny for $ip"
}

record_block() {
  local ip="$1"; local expires=$(( $(date +%s) + BLOCK_DURATION ))

unblock_expired() {
  local now ts ip
  now=$(date +%s)
  local tmp
  tmp="$(mktemp)"
  if [ -f "$STATE_DIR/blocked.db" ]; then
    while read -r ip ts; do
      if [ -z "$ip" ]; then continue; fi
      if [ "$ts" -le "$now" ]; then
        if command -v ufw &>/dev/null; then
          # remove rule by matching comment; ufw delete requires exact rule line; perform best-effort
          sudo ufw status numbered | tac | nl -ba -w1 -s: | while IFS=: read -r num line; do
            if echo "$line" | grep -q "$ip" && echo "$line" | grep -q "$UFW_COMMENT"; then
              sudo ufw --force delete "$num" || true
            fi
          done
        fi
        log "Unblocked $ip (expired)"
      else
        echo "$ip $ts" >> "$tmp"
      fi
    done < "$STATE_DIR/blocked.db"
    mv -f "$tmp" "$STATE_DIR/blocked.db"
  fi
}

block_ip_temporarily() {

      unblock_expired

      sleep 10
    done

  log "Rollback complete"

  chmod 600 "$tmpkey" "$tmpcrt"
  mv -f "$tmpkey" "$finalkey"
  mv -f "$tmpcrt" "$finalcrt"
  ln -sf "$finalkey" "$CURRENT_KEY_SYM"
  ln -sf "$finalcrt" "$CURRENT_CRT_SYM"
  log "Generated cert pair server-$ts"
}

safe_backup_hosts() {
  local meta_hosts="/etc/hosts.meta"
  local backup_dir="/etc/hosts.meta.bak.d"
  sudo mkdir -p "$backup_dir"
  if [ -f "$meta_hosts" ]; then
    sudo cp -a "$meta_hosts" "$backup_dir/hosts.meta.bak.$(date +%s)" || true
    log "Backed up $meta_hosts"
  fi
  local tmp
  tmp="$(mktemp)"
  for h in "${BLACKHOLES[@]}"; do
    echo "127.0.0.1 $h" >> "$tmp"
  done
  sudo mv "$tmp" "$meta_hosts"
  sudo chmod 644 "$meta_hosts"
  log "Wrote $meta_hosts (sinkholes). Ensure resolver includes it."
}

ufw_deny_idempotent() {
  local ip="$1"
  if ! command -v ufw &>/dev/null; then return 1; fi
  if sudo ufw status | grep -E "DENY.+from $ip" &>/dev/null; then
    log "ufw already denies $ip"
    return 0
  fi
  sudo ufw insert 1 deny from "$ip" to any comment "$UFW_COMMENT" || sudo ufw deny from "$ip" to any comment "$UFW_COMMENT"
  log "Added ufw deny for $ip"
}

record_block() {
  local ip="$1"; local expires=$(( $(date +%s) + BLOCK_DURATION ))
  echo "$ip $expires" >> "$STATE_DIR/blocked.db"
}

unblock_expired() {
  local now ip ts tmp
  now=$(date +%s)
  tmp="$(mktemp)"
  if [ -f "$STATE_DIR/blocked.db" ]; then
    while read -r ip ts; do
      [ -z "$ip" ] && continue
      if [ "$ts" -le "$now" ]; then
        if command -v ufw &>/dev/null; then
          # best-effort: remove rules containing IP and comment
          sudo ufw status numbered | awk -v ip="$ip" -v c="$UFW_COMMENT" '
            /---/ {next}
            {print NR ":" $0}
          ' | while IFS=: read -r num line; do
            if echo "$line" | grep -q "$ip" && echo "$line" | grep -q "$c"; then
              sudo ufw --force delete "$num" || true
            fi
          done || true
        fi
        log "Unblocked $ip (expired)"
      else
        echo "$ip $ts" >> "$tmp"
      fi
    done < "$STATE_DIR/blocked.db"
    mv -f "$tmp" "$STATE_DIR/blocked.db"
  fi
}

block_ip_temporarily() {
  local ip="$1"
  if is_admin_ip "$ip"; then
    log "Refusing to block admin IP $ip"
    return
  fi
  if [ "$FIREWALL_ALERT" != "true" ]; then return; fi
  ufw_deny_idempotent "$ip" && record_block "$ip"
  alert "Blocked $ip for $BLOCK_DURATION seconds due to unauthorized activity"
}

start_node_server() {
  local nodefile="/usr/local/lib/meta/meta-real-tls.js"
  if [ ! -f "$nodefile" ]; then
    cat > "$nodefile" <<'NODEJS'
const https = require('https');
const fs = require('fs');
const { exec } = require('child_process');

function now(){ return new Date().toISOString(); }
function log(s){ console.log(`[META-TLS] ${now()} ${s}`); }

const keyPath = process.env.META_KEY || '/var/lib/meta-fortress/certs/current.key';
const crtPath = process.env.META_CRT || '/var/lib/meta-fortress/certs/current.crt';
const validator = process.env.KEYCLOAK_VALIDATOR_URL || 'https://keycloak.local/validate';

const options = {
  key: fs.readFileSync(keyPath),
  cert: fs.readFileSync(crtPath)
};

https.createServer(options, (req, res) => {
  const clientIP = req.socket.remoteAddress.replace('::ffff:', '');
  log(`Handshake from ${clientIP}`);

  exec(`curl -s -o /dev/null -w "%{http_code}" ${validator}`, (err, stdout, stderr) => {
    if (err) {
      log(`Keycloak validation error`);
    } else {
      log(`Keycloak status ${stdout.trim()}`);
    }
  });

  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ ok: true, timestamp: Date.now(), ip: clientIP }));
}).listen(9898, () => log('Live on 9898 with TLS'));
NODEJS
    sudo chown root:root "$nodefile"
    sudo chmod 644 "$nodefile"
  fi

  if id -u meta-fortress &>/dev/null; then
    sudo -u meta-fortress env META_KEY="$CURRENT_KEY_SYM" META_CRT="$CURRENT_CRT_SYM" KEYCLOAK_VALIDATOR_URL="$KEYCLOAK_VALIDATOR_URL" node "$nodefile" >> "$LOGFILE" 2>&1 &
  else
    env META_KEY="$CURRENT_KEY_SYM" META_CRT="$CURRENT_CRT_SYM" KEYCLOAK_VALIDATOR_URL="$KEYCLOAK_VALIDATOR_URL" node "$nodefile" >> "$LOGFILE" 2>&1 &
  fi
  log "Node TLS server started (PID $!)"
}

watcher_loop() {
  log "Starting watcher loop"
  (
    while true; do
      rotate_logs
      unblock_expired
      for port in "${COMMON_PORTS[@]}"; do
        ss -tn sport = :$port state established 2>/dev/null | awk 'NR>1{print $5}' | sed -E 's/:.*$//' | sort -u | while read -r ip; do
          [ -z "$ip" ] && continue
          if [[ "$ip" =~ ^127\. ]] || [[ "$ip" == "::1" ]]; then continue; fi
          if is_admin_ip "$ip"; then
            log "Allowlisted IP $ip connected to port $port"
            continue
          fi
          alert "Port $port active by $ip"
          block_ip_temporarily "$ip"
        done
      done
      sleep 10
    done
  ) &
  echo $! > "$WATCHER_PID_FILE"
}

verify_fortress() {
  log "Verifying TLS 9898"
  ss -tnlp | grep 9898 || log "Port 9898 not active"
}

rollback_fortress() {
  log "Rollback initiated"
  if [ -f "$WATCHER_PID_FILE" ]; then
    kill "$(cat $WATCHER_PID_FILE)" 2>/dev/null || true
    rm -f "$WATCHER_PID_FILE"
  fi
  if sudo ls /etc/hosts.meta.bak.d/* &>/dev/null; then
    sudo cp -a /etc/hosts.meta.bak.d/* /etc/hosts.meta || true
  fi
  log "Rollback complete"
}

case "${1:-}" in
  start)
    rotate_logs
    generate_cert_atomic
    safe_backup_hosts
    start_node_server
    watcher_loop
    ;;
  rotate-certs)
    generate_cert_atomic
    if systemctl is-active --quiet meta-fortress.service; then
      systemctl reload-or-restart meta-fortress.service || true
    fi
    ;;
  verify) verify_fortress ;;
  rollback) rollback_fortress ;;
  *) echo "Usage: $0 {start|rotate-certs|verify|rollback}"; exit 1 ;;
esac
