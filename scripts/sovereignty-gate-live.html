"""
Audit: undo/redo ledger extension.

Adds reversible-state tracking on top of AuditLedger without changing
its append-only guarantee. Every undo/redo is itself a new, signed
ledger entry -- nothing is ever removed from AuditLedger's history.
This satisfies ARCHITECTURE_RULES.md rule 6 ("Protected operations
produce evidence"): reverting a mistake is itself a protected operation.
"""
from __future__ import annotations
from typing import Any
from sia.audit.ledger import AuditLedger, LedgerEntry
from sia.errors import codes
from sia.errors.exceptions import SIAError
from sia.utils.hashing import hash_object


class UndoRedoError(SIAError):
    def __init__(self, code: str, message: str) -> None:
        super().__init__(code, message)


class UndoableLedger:
    """Content-addressed undo/redo on top of an existing AuditLedger."""

    def __init__(self, ledger: AuditLedger) -> None:
        self._ledger = ledger
        self._snapshots: dict[str, dict[str, Any]] = {}

    @property
    def ledger(self) -> AuditLedger:
        return self._ledger

    def _store(self, content: dict[str, Any] | None) -> str | None:
        if content is None:
            return None
        h = hash_object(content)
        self._snapshots.setdefault(h, dict(content))
        return h

    def _read(self, content_hash: str | None) -> dict[str, Any] | None:
        if content_hash is None:
            return None
        if content_hash not in self._snapshots:
            raise UndoRedoError(
                codes.E_UNDO_NO_PRIOR_STATE,
                f"No snapshot stored for hash {content_hash}",
            )
        return dict(self._snapshots[content_hash])

    def record_state_change(
        self,
        *,
        target: str,
        actor_id: str,
        before: dict[str, Any] | None,
        after: dict[str, Any] | None,
        event_type: str,
        note: str = "",
    ) -> LedgerEntry:
        before_hash = self._store(before)
        after_hash = self._store(after)
        return self._ledger.append(
            event_type,
            {"target": target, "before_hash": before_hash,
             "after_hash": after_hash, "note": note},
            actor_id=actor_id,
        )

    def undo(self, entry_id: int, *, actor_id: str) -> LedgerEntry:
        """Append a *.reverted entry -- never deletes original."""
        entry = self._entry_at(entry_id)
        payload = entry.payload
        target = payload["target"]
        current = self.current_state(target)
        restored = self._read(payload["before_hash"])
        return self.record_state_change(
            target=target, actor_id=actor_id,
            before=current, after=restored,
            event_type=f"{entry.event_type}.reverted",
            note=f"Undo of entry {entry_id} on {target}",
        )

    def redo(self, entry_id: int, *, actor_id: str) -> LedgerEntry:
        """Append a *.reapplied entry."""
        entry = self._entry_at(entry_id)
        payload = entry.payload
        if payload["after_hash"] is None:
            raise UndoRedoError(codes.E_REDO_NO_NEXT_STATE,
                                f"Entry {entry_id} has no after state")
        target = payload["target"]
        current = self.current_state(target)
        restored = self._read(payload["after_hash"])
        return self.record_state_change(
            target=target, actor_id=actor_id,
            before=current, after=restored,
            event_type=f"{entry.event_type}.reapplied",
            note=f"Redo of entry {entry_id} on {target}",
        )

    def current_state(self, target: str) -> dict[str, Any] | None:
        for entry in reversed(self._ledger.all_entries()):
            if entry.payload.get("target") == target:
                return self._read(entry.payload["after_hash"])
        return None

    def history_for(self, target: str) -> list[LedgerEntry]:
        return [e for e in self._ledger.all_entries()
                if e.payload.get("target") == target]

    def _entry_at(self, entry_id: int) -> LedgerEntry:
        entries = self._ledger.all_entries()
        if entry_id < 0 or entry_id >= len(entries):
            raise UndoRedoError(codes.E_UNDO_NO_ENTRY,
                                f"No entry with id {entry_id}")
        entry = entries[entry_id]
        if "target" not in entry.payload:
            raise UndoRedoError(codes.E_UNDO_NOT_UNDOABLE,
                                f"Entry {entry_id} has no undoable target")
        return entry

    def record_reasoning_trace(
        self,
        actor_id: str,
        model_id: str,
        *,
        summary: str,
        content_hash: str,
        decision: str | None = None,
        related_entry_id: int | None = None,
        timestamp: str | None = None,
    ) -> None:
        """Record that a model reasoned before acting -- locally, on-device.

        Only summary (short, human-written) and content_hash (SHA3-512
        of the full reasoning text) enter the ledger. The full reasoning
        text belongs in a local-only store (e.g. UndoableLedger snapshot)
        and must never appear in any export to GitHub or external storage.

        Raises SIAError if summary or content_hash is empty -- a reasoning
        trace with nothing to verify against is not evidence.
        """
        if not summary or not content_hash:
            raise SIAError(
                codes.E_REASONING_TRACE_INVALID,
                "record_reasoning_trace requires summary + content_hash",
            )
        self._ledger.append(
            "reasoning.trace",
            {
                "model_id": model_id,
                "summary": summary,
                "content_hash": content_hash,
                "decision": decision,
                "related_entry_id": related_entry_id,
            },
            actor_id=actor_id,
            timestamp=timestamp,
        )

from __future__ import annotations
import pytest
from sia.audit.ledger import AuditLedger
from sia.audit.undo_redo import UndoableLedger, UndoRedoError

def make_undoable():
    return UndoableLedger(AuditLedger())

def test_record_state_change_is_undoable_and_chain_valid():
    u = make_undoable()
    u.record_state_change(target="policy/allowlist.json",
        actor_id="user:derek", before=None, after={"allow":["anthropic"]},
        event_type="policy.updated", note="init")
    u.record_state_change(target="policy/allowlist.json",
        actor_id="model:claude",
        before={"allow":["anthropic"]},
        after={"allow":["anthropic","openai"]},
        event_type="policy.updated", note="added openai")
    assert u.current_state("policy/allowlist.json") == {"allow":["anthropic","openai"]}
    u.ledger.verify_chain()

def test_undo_restores_prior_state_without_deleting_history():
    u = make_undoable()
    u.record_state_change(target="f", actor_id="user:derek",
        before=None, after={"v":1}, event_type="f.updated")
    u.record_state_change(target="f", actor_id="model:grok",
        before={"v":1}, after={"v":2},
        event_type="f.updated", note="wrong change")
    assert u.current_state("f") == {"v":2}
    u.undo(1, actor_id="user:derek")
    assert u.current_state("f") == {"v":1}
    assert len(u.ledger) == 3  # create, modify, undo all remain
    u.ledger.verify_chain()

def test_redo_reapplies_the_undone_change():
    u = make_undoable()
    u.record_state_change(target="f", actor_id="user:derek",
        before=None, after={"v":1}, event_type="f.updated")
    u.record_state_change(target="f", actor_id="model:claude",
        before={"v":1}, after={"v":2}, event_type="f.updated")
    u.undo(1, actor_id="user:derek")
    u.redo(1, actor_id="user:derek")
    assert u.current_state("f") == {"v":2}
    u.ledger.verify_chain()

def test_undo_unknown_entry_raises():
    u = make_undoable()
    with pytest.raises(UndoRedoError): u.undo(0, actor_id="user:derek")

def test_undo_non_state_change_entry_raises():
    u = make_undoable()
    u.ledger.append("authority.init", {"version":"0.1.0"}, actor_id="user:derek")
    with pytest.raises(UndoRedoError): u.undo(0, actor_id="user:derek")

def test_history_for_returns_only_matching_target():
    u = make_undoable()
    u.record_state_change(target="a", actor_id="user:derek",
        before=None, after={"v":1}, event_type="x")
    u.record_state_change(target="b", actor_id="user:derek",
        before=None, after={"v":1}, event_type="x")
    u.record_state_change(target="a", actor_id="user:derek",
        before={"v":1}, after={"v":2}, event_type="x")
    assert len(u.history_for("a")) == 2
    assert len(u.history_for("b")) == 1

from __future__ import annotations
import pytest
from sia.audit.ledger import AuditLedger
from sia.audit.recorder import AuditRecorder
from sia.errors.exceptions import SIAError
from sia.utils.hashing import hash_object

def test_reasoning_trace_records_hash_not_full_text():
    ledger = AuditLedger()
    recorder = AuditRecorder(ledger)
    full = "step 1: check policy. step 2: check quota. step 3: proceed."
    digest = hash_object({"text": full})
    recorder.record_reasoning_trace(
        actor_id="model:claude", model_id="anthropic.claude",
        summary="checked policy and quota before proceeding",
        content_hash=digest, decision="proceed")
    entry = ledger.tail(1)[0]
    assert entry.event_type == "reasoning.trace"
    assert entry.payload["content_hash"] == digest
    # Full reasoning text must never appear in the ledger payload
    assert full not in str(entry.payload)
    ledger.verify_chain()

def test_reasoning_trace_requires_summary_and_hash():
    recorder = AuditRecorder(AuditLedger())
    with pytest.raises(SIAError):
        recorder.record_reasoning_trace(actor_id="model:claude",
            model_id="anthropic.claude", summary="", content_hash="abc")
    with pytest.raises(SIAError):
        recorder.record_reasoning_trace(actor_id="model:claude",
            model_id="anthropic.claude", summary="did something", content_hash="")

def test_reasoning_trace_can_reference_a_related_ledger_entry():
    ledger = AuditLedger()
    recorder = AuditRecorder(ledger)
    recorder.record_authority_init(actor_id="user:derek", version="0.1.0")
    recorder.record_reasoning_trace(actor_id="model:grok", model_id="xai.grok",
        summary="justification for init above",
        content_hash=hash_object({"text":"..."}), related_entry_id=0)
    assert ledger.tail(1)[0].payload["related_entry_id"] == 0
